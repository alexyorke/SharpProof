#!/usr/bin/env bash
set -euo pipefail
repo=/workspace/SharpProof
task_root=$(mktemp -d /tmp/sharpproof-const-census-controls.XXXXXXXX)
cp "$repo/artifacts/immutability-const-semantic/ConstAudit.csproj" "$task_root/ConstAudit.csproj"
cp "$repo/artifacts/immutability-const-semantic/ConstAudit.cs" "$task_root/ConstAudit.cs"
cp "$repo/artifacts/immutability-const-semantic/ResidualAudit.cs" "$task_root/ResidualAudit.cs"
dotnet build "$task_root/ConstAudit.csproj" -c Release --verbosity quiet
mkdir -p "$task_root/control" "$task_root/results"
export_results() {
 mkdir -p "$repo/artifacts/immutability-const-semantic/binding-control-results"
 for result in "$task_root/results/"*.tsv; do
  test -e "$result" && cp "$result" "$repo/artifacts/immutability-const-semantic/binding-control-results/"
 done
}
trap export_results EXIT
cat > "$task_root/control/Control.csproj" <<'XML'
<Project Sdk="Microsoft.NET.Sdk"><PropertyGroup><OutputType>Exe</OutputType><TargetFramework>net10.0</TargetFramework><LangVersion>12</LangVersion><Nullable>enable</Nullable><EnableNETAnalyzers>false</EnableNETAnalyzers><EnableDefaultCompileItems>false</EnableDefaultCompileItems></PropertyGroup><ItemGroup><Compile Include="../control/../control/Control.cs" /></ItemGroup></Project>
XML
cat > "$task_root/control/Control.cs" <<'CS'

using System.Linq;

printf 'Control.cs\n' > "$task_root/universe.txt"
cat > "$task_root/inject.targets" <<'XML'
<Project><ItemGroup><CompilerVisibleProperty Include="ConstAuditOutput;ConstAuditRoot;ConstAuditUniverse;ConstAuditVerify;MSBuildProjectFullPath;TargetFramework;Configuration"/><Analyzer Include="$(ConstAuditAnalyzer)"/></ItemGroup></Project>
XML
dotnet build "$task_root/control/Control.csproj" -c Release --verbosity quiet -p:CustomAfterMicrosoftCommonTargets="$task_root/inject.targets" -p:ConstAuditAnalyzer="$task_root/bin/Release/netstandard2.0/ConstAudit.dll" -p:ConstAuditOutput="$task_root/results" -p:ConstAuditRoot="$task_root/control" -p:ConstAuditUniverse="$task_root/universe.txt" -p:ConstAuditVerify=true
pwsh -NoLogo -NoProfile -File "$repo/artifacts/immutability-const-semantic/check-controls.ps1" -OutputRoot "$task_root/results"
mkdir -p "$repo/artifacts/immutability-const-semantic/binding-control-results"
cp "$task_root/results/"*.tsv "$repo/artifacts/immutability-const-semantic/binding-control-results/"
printf 'Semantic const census representative: %s\n' "$task_root"
