param([Parameter(Mandatory=$true)][string]$OutputRoot)
$ErrorActionPreference='Stop'
$contexts=[Collections.Immutable.ImmutableDictionary]::CreateBuilder[string,object]([StringComparer]::Ordinal)
$generated=0;$nonempty=0
foreach($report in Get-ChildItem -LiteralPath $OutputRoot -Filter '*.tsv'){
 $paths=@(Import-Csv ([IO.Path]::ChangeExtension($report.FullName,'.paths.txt')) -Delimiter ([char]9))
 if($paths.Count -eq 0){throw 'Compiler report has no path authority'}
 $ids=@($paths.ContextSha256|Sort-Object -Unique)
 if($ids.Count -ne 1 -or $ids[0] -cnotmatch '^[0-9a-f]{64}$'){throw 'Missing or ambiguous compiler path context'}
 $id=$ids[0];if($contexts.ContainsKey($id)){throw 'Duplicate compilation context'}
 $units=@(Import-Csv $report.FullName -Delimiter ([char]9)|Where-Object Kind -eq 'Unit')
 if($units.Count -eq 0){if(@($paths|Where-Object Tracked -eq 'True').Count){throw 'Empty report contains tracked compiler paths'};foreach($path in $paths){if($path.ResolvedPath -notmatch '(^|/)obj/' -or $path.ResolvedPath -notmatch '(\.g\.cs|\.AssemblyInfo\.cs|\.AssemblyAttributes\.cs)$'){throw 'Zero-unit report contains an unexplained physical source'}};$generated++}else{$nonempty++;if(@($units|Where-Object ContextSha256 -cne $id).Count){throw 'Compiler unit/path context mismatch'}}
 $contexts.Add($id,$paths)
}
if($contexts.Count -eq 0){throw 'No actual compiler contexts'}
foreach($kind in 'references','types','residual'){
 $reports=@(Get-ChildItem (Join-Path $OutputRoot $kind) -Filter '*.tsv')
 if($reports.Count -ne $contexts.Count){throw "Missing $kind reports"}
 foreach($report in $reports){$id=$report.BaseName.Substring($kind.Length+1);if(-not $contexts.ContainsKey($id)){throw "Unknown $kind context"};$rows=@(Import-Csv $report.FullName -Delimiter ([char]9));if(@($rows|Where-Object ContextSha256 -cne $id).Count){throw "Wrong $kind filename context"}}
}
$referenceGaps=[Collections.Immutable.ImmutableArray]::CreateBuilder[string]();$typeCount=0;$joined=0;$unbound=0
foreach($id in $contexts.Keys){
 $types=@(Import-Csv (Join-Path $OutputRoot "types/types-$id.tsv") -Delimiter ([char]9));$typeMap=[Collections.Immutable.ImmutableDictionary]::CreateBuilder[string,object]([StringComparer]::Ordinal)
 foreach($row in $types){
  $expected=[Convert]::ToHexString([Security.Cryptography.SHA256]::HashData([Text.Encoding]::UTF8.GetBytes($row.SymbolIdentity))).ToLowerInvariant()
  if($row.TypeId -cne $expected -or $typeMap.ContainsKey($row.TypeId)){throw 'Type identity hash/uniqueness failure'}
  $typeMap.Add($row.TypeId,$row);$typeCount++
  if($row.Origin -eq 'Source'){foreach($path in $row.SourcePaths.Split(';')){if($path -cnotin $contexts[$id].ResolvedPath){throw "Source type lacks compiler path authority: $path"}}}elseif($row.SourcePaths -ne ''){throw 'Metadata type has source paths'}
 }
 $residuals=@(Import-Csv (Join-Path $OutputRoot "residual/residual-$id.tsv") -Delimiter ([char]9))
 foreach($row in $residuals){if($row.Kind -eq 'Unit'){continue};if($row.Type -eq '<unbound>'){if($row.TypeId -ne '' -or $row.AssemblyIdentity -ne ''){throw 'False unbound type identity'};$unbound++;continue};if(-not $typeMap.ContainsKey($row.TypeId)){throw 'Residual missing exact type join'};$type=$typeMap[$row.TypeId];if($type.Type -cne $row.Type -or $type.AssemblyIdentity -cne $row.AssemblyIdentity -or $type.SymbolIdentity -cne $row.SymbolIdentity){throw 'Residual type authority mismatch'};$joined++}
 foreach($row in Import-Csv (Join-Path $OutputRoot "references/references-$id.tsv") -Delimiter ([char]9)){
  $aliases=@(if($row.Aliases -ne ''){$row.Aliases.Split(';')});$sorted=[string[]]$aliases.Clone();[Array]::Sort($sorted,[StringComparer]::Ordinal);if(($sorted -join ';') -cne $row.Aliases){throw 'Noncanonical metadata aliases'}
  if($row.Authority -eq 'ExactMetadataFileSha256'){if($row.ContentSha256 -cnotmatch '^[0-9a-f]{64}$'){throw 'Invalid exact metadata SHA256'}}elseif($row.Authority -in 'MissingMetadataFile','UnhashedMetadataReference'){if($row.ContentSha256 -ne ''){throw 'False metadata hash authority'};$referenceGaps.Add($id+[char]9+$row.ReferencePath+[char]9+$row.Authority)}else{throw 'Unknown metadata authority'}
 }
}
[IO.File]::WriteAllText((Join-Path $OutputRoot 'metadata-authority-gaps.txt'),($referenceGaps.ToImmutable()-join [char]10),[Text.UTF8Encoding]::new($false))
Write-Output "Type/reference authority: $($contexts.Count) contexts ($nonempty tracked, $generated generated-only), $typeCount types, $joined exact residual joins, $unbound explicit unbound observations, $($referenceGaps.Count) metadata authority gaps."
if($referenceGaps.Count){throw 'Final compiler census has unauthenticated metadata references'}
