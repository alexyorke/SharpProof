#!/usr/bin/env bash
set -euo pipefail
repo=/workspace/SharpProof
task_root=$(mktemp -d /tmp/sharpproof-residual-controls.XXXXXXXX)
mkdir -p "$task_root/tool" "$task_root/sample" "$task_root/before" "$task_root/after"
cp "$repo/artifacts/immutability-const-semantic/ConstAudit.csproj" "$task_root/tool/"
cp "$repo/artifacts/immutability-const-semantic/ConstAudit.cs" "$task_root/tool/"
cp "$repo/artifacts/immutability-const-semantic/ResidualAudit.cs" "$task_root/tool/"
dotnet build "$task_root/tool/ConstAudit.csproj" -c Release --verbosity quiet
mkdir -p "$task_root/reference"
cat > "$task_root/reference/ReferenceFixture.csproj" <<'XML'
<Project Sdk="Microsoft.NET.Sdk"><PropertyGroup><TargetFramework>net10.0</TargetFramework><EnableNETAnalyzers>false</EnableNETAnalyzers></PropertyGroup></Project>
XML
printf '%s\n' 'namespace ReferenceFixture; public sealed class Marker { public const int Value = 1; }' > "$task_root/reference/Marker.cs"
dotnet build "$task_root/reference/ReferenceFixture.csproj" -c Release --verbosity quiet
cat > "$task_root/sample/Sample.csproj" <<'XML'
<Project Sdk="Microsoft.NET.Sdk"><PropertyGroup><TargetFramework>net10.0</TargetFramework><LangVersion>12</LangVersion><Nullable>enable</Nullable><EnableNETAnalyzers>false</EnableNETAnalyzers><ImplicitUsings>enable</ImplicitUsings></PropertyGroup><ItemGroup><Reference Include="ReferenceFixture"><HintPath>../reference/bin/Release/net10.0/ReferenceFixture.dll</HintPath></Reference></ItemGroup></Project>
XML
cat > "$task_root/sample/Cases.cs" <<'CS'
using System.Collections.Immutable;
public sealed class Cases(int[] captured)
{
    public int[] Capture => captured;
    public List<int> Values { get; set; } = new();
    public static T Generic<T>(T input) => input;
    public static T OtherGeneric<T>(T input) => input;
    public static Cases Create() => new([]);
    public static FrozenCase Freeze() => new(ImmutableArray<int>.Empty);
    public static int ReferenceValue() => ReferenceFixture.Marker.Value;
    public static ReferenceFixture.Marker MetadataRecord() => new();
    public static object DynamicCall(dynamic target) => target.Unknown();
    public static int Read(ReadOnlySpan<int> input) => input.Length;
    public static int[] Make(IReadOnlyList<int[]> input)
    {
        GC.KeepAlive(nameof(input));
        var inferred = File.ReadAllBytes("unused");
        var builder = ImmutableArray.CreateBuilder<int>();
        builder.Add(inferred.Length);
        var sequence = ImmutableArray<int>.Empty;
        sequence = sequence.Add(builder.Count);
        sequence.Add(-1);
        { var duplicate = Array.Empty<int>(); GC.KeepAlive(duplicate); }
        { var duplicate = Array.Empty<int>(); GC.KeepAlive(duplicate); }
        int[] Local() => input[0];
        GC.KeepAlive(sequence);
        return Local();
    }
}
public sealed record FrozenCase(ImmutableArray<int> Values);
CS
printf '%s\n' 'public static class Zero { public const int Value = 1; }' > "$task_root/sample/Zero.cs"
printf '%s\n' 'Cases.cs' 'Zero.cs' > "$task_root/universe.txt"
cat > "$task_root/inject.targets" <<'XML'
<Project><ItemGroup><CompilerVisibleProperty Include="ConstAuditOutput;ConstAuditRoot;ConstAuditUniverse;ConstAuditVerify;MSBuildProjectFullPath;TargetFramework;Configuration"/><Analyzer Include="$(ConstAuditAnalyzer)"/></ItemGroup></Project>
XML
set -- -c Release --verbosity quiet /t:Rebuild "-p:CustomAfterMicrosoftCommonTargets=$task_root/inject.targets" "-p:ConstAuditAnalyzer=$task_root/tool/bin/Release/netstandard2.0/ConstAudit.dll" "-p:ConstAuditRoot=$task_root/sample" "-p:ConstAuditUniverse=$task_root/universe.txt" -p:ConstAuditVerify=false
dotnet build "$task_root/sample/Sample.csproj" "$@" "-p:ConstAuditOutput=$task_root/before"
sed -i '1i// Coordinate-only shift must preserve nonambiguous root identities.' "$task_root/sample/Cases.cs"
dotnet build "$task_root/sample/Sample.csproj" "$@" "-p:ConstAuditOutput=$task_root/after"
pwsh -NoLogo -NoProfile -File "$repo/artifacts/immutability-const-semantic/check-residual-controls.ps1" -Before "$task_root/before" -After "$task_root/after"
sed -i 's/Value = 1/Value = 2/' "$task_root/reference/Marker.cs"
dotnet build "$task_root/reference/ReferenceFixture.csproj" -c Release --verbosity quiet
dotnet build "$task_root/sample/Sample.csproj" "$@" "-p:ConstAuditOutput=$task_root/metadata-changed"
pwsh -NoLogo -NoProfile -File "$repo/artifacts/immutability-const-semantic/check-type-reference-controls.ps1" -Before "$task_root/after" -After "$task_root/metadata-changed"
pwsh -NoLogo -NoProfile -File "$repo/artifacts/immutability-const-semantic/check-authority-negative-controls.ps1" -Baseline "$task_root/after"
mkdir -p "$task_root/generated"
printf '%s\n' '<Project Sdk="Microsoft.NET.Sdk"><PropertyGroup><TargetFramework>net10.0</TargetFramework><ImplicitUsings>enable</ImplicitUsings><EnableNETAnalyzers>false</EnableNETAnalyzers></PropertyGroup></Project>' > "$task_root/generated/GeneratedOnly.csproj"
: > "$task_root/generated-universe.txt"
dotnet build "$task_root/generated/GeneratedOnly.csproj" -c Release --verbosity quiet \
 "-p:CustomAfterMicrosoftCommonTargets=$task_root/inject.targets" "-p:ConstAuditAnalyzer=$task_root/tool/bin/Release/netstandard2.0/ConstAudit.dll" \
 "-p:ConstAuditRoot=$task_root/generated" "-p:ConstAuditUniverse=$task_root/generated-universe.txt" "-p:ConstAuditOutput=$task_root/generated-results" -p:ConstAuditVerify=false
pwsh -NoLogo -NoProfile -File "$repo/artifacts/immutability-const-semantic/check-type-reference-authority.ps1" -OutputRoot "$task_root/generated-results"
destination="$repo/artifacts/immutability-const-semantic/residual-control-results-$(basename "$task_root")"
mkdir -p "$destination"
cp -a "$task_root/before" "$task_root/after" "$task_root/metadata-changed" "$task_root/generated-results" "$destination/"
printf 'Semantic residual controls: %s\n' "$task_root"
