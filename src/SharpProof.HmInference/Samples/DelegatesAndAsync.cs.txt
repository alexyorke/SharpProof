#nullable enable
using System;
using System.Threading.Tasks;

public delegate R SampleMap<A,R>(A value);
public sealed class SampleBox<T>
{
    public T Value;
    public SampleBox(T value) { Value=value; }
    public T Get() => Value;
}
public static class DelegateAsyncSamples
{
    public static void Forward(Action<int> sink,int value) => sink(value);
    public static Action Remember(int value) => delegate { Ignore(value); };
    private static void Ignore<T>(T value) {}
    public static Func<Task<int>> Capture(int value) => async () => { await Task.Yield(); return value; };
    public static Func<ValueTask<int>> ValueCapture(int value) => async () => { await Task.Yield(); return value; };
    public static async void Fire(Task<int> task,Action<int> done) { done(await task); }
    public static SampleMap<int,int> Custom(int seed) => value => Pick(seed,value);
    private static T Pick<T>(T first,T second) => first;
    public static Func<int> Read(SampleBox<int> box) => box.Get;
}
