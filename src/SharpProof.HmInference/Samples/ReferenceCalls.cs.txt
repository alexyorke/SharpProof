#nullable enable
public delegate T SampleRefRead<T>(ref T value);
public delegate ref T SampleRefBorrow<T>(ref T value);
public static class SampleRefApi
{
    public static T Read<T>(ref T value) => value;
    public static void Store<T>(ref T target, T value) => target = value;
    public static void Copy<T>(T value, out T result) => result = value;
    public static ref T Borrow<T>(ref T value) => ref value;
}
public static class ReferenceCallSamples
{
    public static int RefRead(int value) => SampleRefApi.Read(ref value);
    public static int RefStore(ref int target, int value)
    {
        SampleRefApi.Store(ref target, value);
        return target;
    }
    public static int OutCopy(int value)
    {
        SampleRefApi.Copy(value, out var result);
        return result;
    }
    public static ref int Borrow(ref int value) => ref SampleRefApi.Borrow(ref value);
    public static int WriteAlias(ref int target, int value)
    {
        ref var alias = ref SampleRefApi.Borrow(ref target);
        alias = value;
        return target;
    }
    public static int DelegateRead(SampleRefRead<int> read, ref int value) => read(ref value);
    public static ref int DelegateBorrow(SampleRefBorrow<int> borrow, ref int value) => ref borrow(ref value);
    public static SampleRefRead<int> RefLambda(bool ignored) => (ref int value) => value;
}
