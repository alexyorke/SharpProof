#nullable enable
using System;
using System.Threading.Tasks;
public sealed class SampleResource<T>(T value):IDisposable,IAsyncDisposable
{
    public T Value=>value;
    public void Dispose() { }
    public ValueTask DisposeAsync()=>default;
}
public static class ResourceSamples
{
    public static int NullableResource(SampleResource<int>? resource,int value)
    {using(resource){return value;}}
    public static int ResourcePayload(SampleResource<int> resource)
    {using var copy=resource;return copy.Value;}
    public static async Task<int> AsyncResourcePayload(SampleResource<int> resource)
    {await using var copy=resource;return copy.Value;}
    public static async Task<int> AsyncResourceContract(IAsyncDisposable resource,int value)
    {await using(resource){return value;}}
}
